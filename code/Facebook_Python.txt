from itertools import groupby

string = "Geeeks fore Geeksee"
letter = "ee"

q = 0
cnt = 0
length = len(string)

while q < length:
    if string[q:length].find(letter) > 0:
        q = q + string[q:length].find(letter) + (len(letter))
        cnt = cnt +1
    else:
        q = q + 1

print(cnt)
